# imageio dispatches build_reel.py's reel.mp4 (imageio.mimwrite(..., codec=
# "libx264")) and reel.gif writes to a plugin. Without a video backend
# installed this does NOT fail with a clear "install a video backend"
# message: depending on what else is on the machine it instead raises either
# a wrong-plugin TypeError, or (if neither imageio-ffmpeg nor av is present)
# an opaque TiffWriter error with no mention of video at all. Install both of
# the packages below — do not substitute one for "imageio[pyav]" and assume
# that alone fixes an mp4-writing failure; imageio's actual production write
# path here goes through imageio-ffmpeg, not pyav.
imageio==2.37.0
imageio-ffmpeg==0.6.0  # ffmpeg backend imageio.mimwrite() needs to write reel.mp4/gif

# --- used only by docs/marketing/tests/ --------------------------------
av==18.0.0             # test_build_reel.py reads reel.mp4 back with
                       # iio.imread(mp4, plugin="pyav") to assert on frame
                       # content/count/order; not used by any script.
pytest==9.0.2          # runs docs/marketing/tests/
